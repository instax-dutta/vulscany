```
// Single export:
export function proxy(request: NextRequest): NextResponse
export const config: { matcher: string[] }

// proxy() behavior:
// 1. Generates nonce from crypto.randomUUID(), base64-encodes it
// 2. Sets Content-Security-Policy header with nonce-based directives
// 3. Sets x-nonce header on request for Next.js rendering
// 4. If pathname starts with /dashboard and no github_token cookie -> redirect to /
// 5. If pathname starts with /api/ (but not /api/auth/) and no github_token cookie -> 401 JSON
```
